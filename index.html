/** * BET LEDGER BRIDGE * Paste this whole file into the Apps Script editor attached to your sheet. * * Change SECRET below to any random string of your own before deploying. * The web app and this script must use the same value. * * SHEET_ID: leave as '' if you opened this from the sheet's own * Extensions > Apps Script menu. If you created the project at * script.google.com instead, paste your sheet's ID here -- it is the long * string in the sheet URL between /d/ and /edit */ const SECRET = 'CHANGE-ME-to-something-random'; const SHEET_ID = ''; const TAB = 'live'; const COLS = ['id','date','channel','sport','type','structure','selection','odds', 'base','combos','stake','timing','result','payout','odds_source','note','logged_at', 'settles']; /** Returns the sheet, creating it with headers if missing. */ function getSheet_() { const ss = SHEET_ID ? SpreadsheetApp.openById(SHEET_ID) : SpreadsheetApp.getActiveSpreadsheet(); if (!ss) throw new Error('No spreadsheet. Set SHEET_ID at the top of this file.'); let sh = ss.getSheetByName(TAB); if (!sh) { sh = ss.insertSheet(TAB); sh.appendRow(COLS); sh.setFrozenRows(1); } if (sh.getLastRow() === 0) { sh.appendRow(COLS); sh.setFrozenRows(1); } return sh; } function ok_(obj) { return json_(Object.assign({ ok: true }, obj)); } function err_(msg) { return json_({ ok: false, error: msg }); } function json_(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } /** GET — read every bet back as JSON. */ function doGet(e) { try { const p = (e && e.parameter) || {}; if (p.secret !== SECRET) return err_('bad secret'); const sh = getSheet_(); if (sh.getLastRow() < 2) return ok_({ bets: [] }); const values = sh.getRange(2, 1, sh.getLastRow() - 1, COLS.length).getValues(); const bets = values .filter(function (r) { return r[0] !== '' && r[0] !== null; }) .map(function (r) { const o = {}; COLS.forEach(function (c, i) { o[c] = r[i]; }); // dates come back as Date objects; send plain YYYY-MM-DD ['date', 'settles'].forEach(function (k) { if (o[k] instanceof Date) { o[k] = Utilities.formatDate(o[k], Session.getScriptTimeZone(), 'yyyy-MM-dd'); } }); return o; }); return ok_({ bets: bets, count: bets.length }); } catch (ex) { return err_(String(ex)); } } /** * POST — add or settle bets. * Body must be JSON sent as text/plain (see note in the setup steps). * * { secret, action: 'add', bets: [ {…}, {…} ] } * { secret, action: 'settle', id, result, payout } */ function doPost(e) { const lock = LockService.getScriptLock(); try { lock.waitLock(20000); const body = JSON.parse(e.postData.contents); if (body.secret !== SECRET) return err_('bad secret'); const sh = getSheet_(); if (body.action === 'add') { const bets = body.bets || []; if (!bets.length) return err_('no bets supplied'); const stamp = new Date().toISOString(); const rows = bets.map(function (b) { b.logged_at = stamp; b.odds_source = 'recorded'; return COLS.map(function (c) { return b[c] == null ? '' : b[c]; }); }); sh.getRange(sh.getLastRow() + 1, 1, rows.length, COLS.length).setValues(rows); return ok_({ added: rows.length }); } if (body.action === 'settle') { const ids = sh.getRange(2, 1, Math.max(sh.getLastRow() - 1, 1), 1).getValues(); for (let i = 0; i < ids.length; i++) { if (String(ids[i][0]) === String(body.id)) { const row = i + 2; sh.getRange(row, COLS.indexOf('result') + 1).setValue(body.result); sh.getRange(row, COLS.indexOf('payout') + 1).setValue(body.payout); return ok_({ settled: body.id, row: row }); } } return err_('id not found: ' + body.id); } return err_('unknown action: ' + body.action); } catch (ex) { return err_(String(ex)); } finally { lock.releaseLock(); } } /** * Run this once from the editor to confirm everything works. * Select selfTest in the function dropdown, press Run, then check the sheet. */ function selfTest() { const sh = getSheet_(); sh.appendRow(['TEST-1', '2026-09-07', 'DK Sportsbook', 'NFL', 'spread', 'straight', 'TEST ROW - delete me', '-110', 5, 1, 5, 'pregame', '', '', 'recorded', '', new Date().toISOString(), '']); Logger.log('Wrote a test row. Sheet now has ' + sh.getLastRow() + ' rows.'); }
